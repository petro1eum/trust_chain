import ast
from decimal import Decimal, localcontext

def calculate(expression):
    if not isinstance(expression, str) or len(expression) > 256:
        raise ValueError("Expression too large")
    tree = ast.parse(expression, mode="eval")
    if sum(1 for _ in ast.walk(tree)) > 64:
        raise ValueError("Too many operations")
    def visit(node, depth=0):
        if depth > 16:
            raise ValueError("Expression too deep")
        if isinstance(node, ast.Constant) and type(node.value) in (int, float):
            value = Decimal(ast.get_source_segment(expression, node))
        elif isinstance(node, ast.UnaryOp) and isinstance(node.op, (ast.UAdd, ast.USub)):
            value = visit(node.operand, depth+1)
            if isinstance(node.op, ast.USub): value = -value
        elif isinstance(node, ast.BinOp):
            a, b = visit(node.left, depth+1), visit(node.right, depth+1)
            if isinstance(node.op, ast.Add): value = a+b
            elif isinstance(node.op, ast.Sub): value = a-b
            elif isinstance(node.op, ast.Mult): value = a*b
            elif isinstance(node.op, ast.Div): value = a/b
            elif isinstance(node.op, ast.Pow) and b == int(b) and 0 <= b <= 12: value = a**int(b)
            else: raise ValueError("Operator forbidden")
        else:
            raise ValueError("Only bounded arithmetic is supported")
        if not value.is_finite() or abs(value) > Decimal("1e12"):
            raise ValueError("Result out of range")
        return value
    with localcontext() as ctx:
        ctx.prec = 30
        return str(visit(tree.body))

2️⃣ Chain of Trust with an optional model call¶

# Define tools for the LLM
tools = [
    {'type': 'function', 'function': {'name': 'get_weather', 'parameters': {'type': 'object', 'properties': {'city': {'type': 'string'}}}}},
    {'type': 'function', 'function': {'name': 'calculate', 'parameters': {'type': 'object', 'properties': {'expr': {'type': 'string'}}}}},
    {'type': 'function', 'function': {'name': 'search', 'parameters': {'type': 'object', 'properties': {'query': {'type': 'string'}}}}}
]

# Tool implementations
def execute_tool(name, args):
    if name == 'get_weather':
        return f"Sunny, 22°C in {args.get('city', 'Unknown')}"
    elif name == 'calculate':
        return calculate(args.get('expr', '0'))
    elif name == 'search':
        return f"Found 5 results for: {args.get('query', '')}"
    return 'Unknown'


def offline_tool_call(name, args):
    return SimpleNamespace(
        function=SimpleNamespace(
            name=name,
            arguments=json.dumps(args),
        )
    )


# Multi-step agent workflow
chain = []
queries = [
    'What is the weather in Tokyo? Use the tool.',
    'Calculate 2 ** 10 using the calculate tool.',
    'Search for AI security using the search tool.'
]
offline_calls = [
    [offline_tool_call('get_weather', {'city': 'Tokyo'})],
    [offline_tool_call('calculate', {'expr': '2 ** 10'})],
    [offline_tool_call('search', {'query': 'AI security'})],
]

print('🤖 Running multi-step agent workflow...\n')

for i, query in enumerate(queries, 1):
    print(f'📤 Step {i}: {query}')

    if client:
        response = client.chat.completions.create(
            model=OPENROUTER_MODEL,
            messages=[{'role': 'user', 'content': query}],
            tools=tools,
            tool_choice='auto',
            max_tokens=100
        )
        tool_calls = response.choices[0].message.tool_calls or []
    else:
        tool_calls = offline_calls[i - 1]

    for tc_call in tool_calls:
        args = json.loads(tc_call.function.arguments)
        result = execute_tool(tc_call.function.name, args)

        # tc.sign() auto-chains from chain HEAD — no manual parent threading.
        # Pass parent_signature=None only if you want to start a NEW root.
        signed = tc.sign(
            tool_id=tc_call.function.name,
            data={'args': args, 'result': result},
        )
        parent_sig = signed.parent_signature
        chain.append(signed)

        print(f'   ✅ {tc_call.function.name}: {result}')
        print(f'   🔏 Signature: {signed.signature[:32]}...')
        if parent_sig:
            print(f'   ⛓️ Linked to parent: {parent_sig[:32]}...')
    print()

print(f'📊 Chain complete: {len(chain)} tool calls')
print(f'✅ All signatures valid: {all(tc.verify(s) for s in chain)}')

assert all(tc.verify(s) for s in chain)
if client is None:

    assert all(chain[i].parent_signature == chain[i-1].signature for i in range(1, len(chain)))

🧾 Export a .tcreceipt — portable, offline-verifiable proof¶

All the signing above lives inside your TrustChain instance. Sometimes you need to hand someone else — a regulator, a partner, an end user — a self-contained artifact they can verify on their own laptop, without your server.

That's .tcreceipt: one JSON file, one drag-and-drop into examples/verify.html, WebCrypto Ed25519 check, no network required.

from pathlib import Path

from trustchain import Receipt, build_receipt

# Take any signed response we produced above — re-use the first one.
# In a real agent, you'd loop over the facts you want to expose.
receipt = build_receipt(
    signed,
    tc.export_public_key(),
    key_id=tc.get_key_id(),
)

out = Path("demo_fact.tcreceipt")
out.write_text(receipt.to_json(indent=2))
print(f"💾 wrote {out} — {out.stat().st_size} bytes")

# Roundtrip: load + verify + pin to a known public key
loaded = Receipt.load(out.read_text())
v = loaded.verify(
    expected_public_key_b64=tc.export_public_key(),
    max_age_seconds=3600,
)
print(f"\n✅ valid       : {v.valid}")
print(f"   sig_ok     : {v.signature_ok}")
print(f"   identity_ok: {v.identity_ok}")
print(f"   errors     : {v.errors or '—'}")
print(f"   warnings   : {v.warnings or '—'}")

# Try it in the browser! Download examples/verify.html from the repo,
# open it locally (file://), drag demo_fact.tcreceipt onto the drop zone.
# You'll get the SAME verdict, no Python, no server.

assert v.valid and v.signature_ok
