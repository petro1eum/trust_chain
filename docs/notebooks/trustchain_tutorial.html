1️⃣ Basic Usage — Signing Tool Responses¶

Every tool response is automatically signed with Ed25519.

# Create TrustChain instance
tc = TrustChain(TrustChainConfig(enable_nonce=False))

# Define a trusted tool

    return {
        "city": city,
        "temperature": 22.5,
        "conditions": "Sunny",
        "timestamp": int(time.time())
    }

# Call the tool - response is automatically signed!
response = get_weather("Moscow")

print(f"📍 City: {response.data['city']}")
print(f"🌡️ Temperature: {response.data['temperature']}°C")
print(f"\n🔐 Signature: {response.signature[:48]}...")
print(f"✅ Verified: {response.is_verified}")

2️⃣ Chain of Trust — Linked Operations¶

Link operations together for complete audit trails.

tc_chain = TrustChain(TrustChainConfig(enable_nonce=False))

@tc_chain.tool("step_processor")
def process_step(step: int, data: str):
    return {"step": step, "processed": data.upper(), "timestamp": time.time()}

# Build a chain of operations using the PUBLIC API — tc.sign().
# It auto-chains via chain HEAD, so we don't need to thread parent_signature
# manually. Pass parent_signature=None for the first link if you want to
# force a fresh root; here we just let auto-chain do its job.
step1 = tc_chain.sign("step1", {"action": "fetch_data", "source": "API"})
step2 = tc_chain.sign("step2", {"action": "transform", "rows": 1000})
step3 = tc_chain.sign("step3", {"action": "store", "destination": "DB"})

3️⃣ Multi-Tenant Isolation¶

Each tenant has isolated keys — signatures from one tenant cannot verify in another.

    await asyncio.sleep(0.01)  # Simulate async work

        "count": len(items),
        "sum": sum(items),
        "avg": sum(items) / len(items) if items else 0

    times = []
    for _ in range(100):
        start = time.perf_counter()
        await process_data([1, 2, 3, 4, 5])

6️⃣ Git-like Versioning & Revert (Undo for AI)¶

TrustChain has a built-in git-like audit and version control layer in the OSS version. Under the hood, every signed operation behaves like a commit with parent links. This enables operators to checkpoint, tag, branch, blame, reset, and revert agent states with ease.

import tempfile

from trustchain import TrustChain, TrustChainConfig

# Create a local file-based TrustChain repository in a temporary folder
with tempfile.TemporaryDirectory() as tmpdir:
    tc_git = TrustChain(TrustChainConfig(
        enable_nonce=False,
        chain_storage="file",
        chain_dir=tmpdir
    ))

    # 1. Sign a few operations (simulating agent actions)
    op1 = tc_git.sign("database_tool", {"action": "INSERT", "user": "alice"})
    print(f"📝 Created op1: {op1.signature_id[:16]}...")

    op2 = tc_git.sign("payment_tool", {"action": "PAY", "amount": 100.00})
    print(f"📝 Created op2: {op2.signature_id[:16]}...")

    # 2. Git-like Version Control API
    # Create a tag, checkpoint and branch
    tc_git.chain.tag("v1.0-release")
    tc_git.chain.checkpoint("payment-checkpoint")
    tc_git.chain.branch("hotfix-branch")

    print("\n📁 Created Git-like references:")
    refs = tc_git.chain.list_refs()
    print("   Tags:", [r["name"] for r in refs["tag"]])
    print("   Checkpoints:", [r["name"] for r in refs["checkpoint"]])
    print("   Branches:", [r["name"] for r in refs["head"]])

    # 3. Forensic Blame
    print("\n🔍 Forensic audit log (blame payment_tool):")
    blamed_ops = tc_git.chain.blame("payment_tool")
    for op in blamed_ops:
        print(f"   [{op['timestamp']}] ID: {op['id']} | Data: {op['data']}")

    # 4. Undo for AI (Compensatory Revert)
    print("\n🔄 Reverting op2 (Undo action):")
    # Resolve the local operation ID of op2 from the log
    op2_id = next(op["id"] for op in tc_git.chain.log() if op["signature"] == op2.signature)
    revert_op = tc_git.revert(op_id=op2_id, reason="User cancelled transaction")
    print(f"   Revert op created: {revert_op.signature_id[:16]}...")
    print(f"   Revert payload: {revert_op.data}")

    # 5. Cryptographic Diffing (Compare Operations)
    print("\n🔍 Diffing op1 and op2:")
    op1_id = next(op["id"] for op in tc_git.chain.log() if op["signature"] == op1.signature)
    diff_res = tc_git.chain.diff(op1_id, op2_id)
    print(f"   Same tool: {diff_res['same_tool']}")
    print(f"   Time delta: {diff_res['time_delta_seconds']:.6f}s")
    print(f"   Operation A: {diff_res['a']['tool']} -> {diff_res['a']['data']}")
    print(f"   Operation B: {diff_res['b']['tool']} -> {diff_res['b']['data']}")

    # 6. Soft Reset (Time-Travel / Rollback)
    print("\n⏳ Resetting HEAD to op1 (soft rollback):")
    reset_res = tc_git.chain.reset(op1_id, soft=True)
    print(f"   New HEAD signature: {reset_res['new_head'][:16]}...")
    print(f"   Detached operations: {reset_res['detached_ids']}")
    print(f"   Current HEAD matches op1: {tc_git.chain.head() == op1.signature}")

🧾 .tcreceipt — Portable Proof-of-Signature (v3.0+)¶

A .tcreceipt is a single JSON file that bundles the signed envelope + public key + (optional) identity. Anyone can verify it offline — Python CLI, JS SDK, or the single-file HTML verifier (examples/verify.html).

The canonical payload is byte-identical across all three verifiers, so a receipt produced in Python verifies in the browser and vice versa.

from trustchain import Receipt, TrustChain, TrustChainConfig, build_receipt

tc = TrustChain(TrustChainConfig(enable_nonce=False))

# Sign a tool fact as usual — think "AI just pulled a record from the DB".
signed = tc.sign("database_query", {"user_id": 42, "balance": 1234.56})
assert tc.verify(signed)

# Build a portable receipt — this is the file you'd hand to a regulator /
# a downstream partner / a skeptical user. No server call, no key exchange.
# key_id пришивается к receipt явно — его используют verifiers для pinning.
receipt = build_receipt(signed, tc.export_public_key(), key_id=tc.get_key_id())

json_blob = receipt.to_json()
print("📄 .tcreceipt size :", len(json_blob), "bytes")
print("🔑 pinned key_id   :", (receipt.key["key_id"] or "")[:16], "...")
print("🔏 algorithm       :", receipt.key["algorithm"])

# Roundtrip: load it back, verify it — no reference to the signing TrustChain instance.
loaded = Receipt.load(json_blob)

# Pin the receipt to a known-good public key (anti-impersonation) and reject
# envelopes older than an hour (anti-replay for short-lived claims).
pinned_pk = tc.export_public_key()
result = loaded.verify(
    max_age_seconds=3600,
    expected_public_key_b64=pinned_pk,
)

print("\n✅ verify:", result.valid)
print("   errors:", result.errors or "—")
print("   warnings:", result.warnings or "—")

# Try the same blob with examples/verify.html — drag & drop, WebCrypto check,
# no installation, works on an airgapped laptop.
